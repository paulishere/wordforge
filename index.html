// ======== SPEECH ========

function getBestVoice(){

  const v=speechSynthesis.getVoices();

  const p=["samantha","google us english","microsoft aria","microsoft jenny","karen","daniel","google uk english female","alex","victoria","microsoft zira"];

  for(const n of p){const f=v.find(x=>x.name.toLowerCase().includes(n)&&x.lang.startsWith("en"));if(f)return f;}

  return v.find(x=>x.lang.startsWith("en-US"))||v.find(x=>x.lang.startsWith("en"))||v[0]||null;

}

function speakCurrentWord(){

  if(!speechSynthesis)return;

  speechSynthesis.cancel();

  const u=new SpeechSynthesisUtterance(state.decodeTarget);

  const v=getBestVoice();if(v)u.voice=v;

  u.rate=0.78;u.pitch=1.05;u.volume=1;

  speechSynthesis.speak(u);

  const btn=document.getElementById('hear-btn');

  btn.style.transform='scale(1.05)';setTimeout(()=>btn.style.transform='',200);

}

function shuffle(a){const b=[...a];for(let i=b.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[b[i],b[j]]=[b[j],b[i]];}return b;}

function pickChallenge(idx){

  const bank=BANK[idx];

  if(!state.usedChallenges[idx])state.usedChallenges[idx]=[];

  let avail=bank.map((_,i)=>i).filter(i=>!state.usedChallenges[idx].includes(i));

  if(avail.length===0){state.usedChallenges[idx]=[];avail=bank.map((_,i)=>i);}

  const pick=avail[Math.floor(Math.random()*avail.length)];

  state.usedChallenges[idx].push(pick);

  return bank[pick];

}

  state.currentBuilding=i;

  state.currentChallenge=pickChallenge(i);

  decodeWordQueue=[...state.currentChallenge.decodeWords];

  showDecodeWord();

}

    'Listen to the word, then tap the letters in order to spell it.'+(decodeWordQueue.length>1?` Word ${decodeWordIndex+1} of ${decodeWordQueue.length}`:'');

}


function handleLetterTap(tile){

  const expected=state.decodeTarget[state.decodeLetterIndex];

  if(tile.dataset.letter===expected){

    tile.classList.add('used');

    const slot=document.getElementById('slot-'+state.decodeLetterIndex);

    slot.textContent=tile.dataset.letter;slot.classList.add('filled');

    state.decodeLetterIndex++;

    if(state.decodeLetterIndex===state.decodeTarget.length){

      for(let i=0;i<state.decodeTarget.length;i++)document.getElementById('slot-'+i).classList.add('correct-flash');

      state.streak++;if(state.streak>state.bestStreak)state.bestStreak=state.streak;

      decodeWordIndex++;

      if(decodeWordIndex<decodeWordQueue.length)setTimeout(()=>showDecodeWord(),1000);

      else setTimeout(()=>showRead(),1000);

    }

  }else{

    tile.style.pointerEvents='none';

    const slot=document.getElementById('slot-'+state.decodeLetterIndex);

    slot.classList.add('wrong-flash');

    setTimeout(()=>{slot.classList.remove('wrong-flash');tile.style.pointerEvents='auto';},500);

  }

}

function handleWordTap(index){

  if(index!==state.currentWordIndex)return;

  const now=Date.now();const words=document.querySelectorAll('.word-token');

  if(index===0)state.readStartTime=now;

  state.readWordTimes.push(now);

  words[index].classList.remove('current');words[index].classList.add('read');

  const progress=((index+1)/words.length)*100;

  document.getElementById('forge-fill').style.width=progress+'%';

  if(index>2){

    const elapsed=(now-state.readStartTime)/60000;const wpm=(index+1)/elapsed;

    updateForgeHeat(wpm);

  }

  state.currentWordIndex++;

  if(state.currentWordIndex<words.length){

    words[state.currentWordIndex].classList.remove('unread');

    words[state.currentWordIndex].classList.add('current');

  }else{setTimeout(()=>showComprehend(),600);}

}


function updateForgeHeat(wpm){

  const label=document.getElementById('forge-label');const fill=document.getElementById('forge-fill');

  const baseline=state.personalBestWPM>0?state.personalBestWPM:70;

  if(wpm>=baseline*1.1){label.textContent='\u{1F525} BLAZING! The forge is white-hot!';fill.style.background='linear-gradient(90deg,#FFD700,#fff,#FFD700)';fill.style.filter='brightness(1.3)';}

  else if(wpm>=baseline*0.85){label.textContent='\u{1F525} Hot! The forge glows bright!';fill.style.background='linear-gradient(90deg,#e94560,#FFD700)';fill.style.filter='brightness(1.1)';}

  else{label.textContent='\u2728 Warming up... keep going!';fill.style.background='linear-gradient(90deg,#CD7F32,#e94560)';fill.style.filter='brightness(1)';}

}

function showComprehend(){

  const c=state.currentChallenge;renderDots('comp-dots',2);

  document.getElementById('comp-question').textContent=c.question;

  const opt=document.getElementById('comp-options');opt.innerHTML='';

  c.answers.forEach((a,i)=>{

    const btn=document.createElement('div');btn.className='comp-option';btn.textContent=a;

    btn.onclick=()=>handleCompAnswer(i,btn);opt.appendChild(btn);

  });

  showScreen('comprehend-screen');

}


function handleCompAnswer(index,btn){

  const c=state.currentChallenge;

  const all=document.querySelectorAll('.comp-option');

  all.forEach(b=>b.style.pointerEvents='none');

  if(index===c.correctAnswer){

    btn.classList.add('correct');state.streak++;

    if(state.streak>state.bestStreak)state.bestStreak=state.streak;

    setTimeout(()=>showReward(),800);

  }else{

    btn.classList.add('incorrect');all[c.correctAnswer].classList.add('correct');

    state.streak=0;

    setTimeout(()=>{all.forEach(b=>{b.classList.remove('correct','incorrect');b.style.pointerEvents='auto';});},1200);

  }

}


// ======== REWARD ========

function showReward(){

  const words=state.currentChallenge.passage.split(/\s+/);

  const totalTime=(state.readWordTimes[state.readWordTimes.length-1]-state.readStartTime)/60000;

  const wpm=Math.round(words.length/totalTime);

  if(wpm>state.personalBestWPM)state.personalBestWPM=wpm;

  const baseline=state.personalBestWPM>0?state.personalBestWPM:70;

  let quality,stars,amt;

  if(wpm>=baseline*0.95){quality='gold';stars=3;amt=3;}

  else if(wpm>=baseline*0.75){quality='silver';stars=2;amt=2;}

  else{quality='bronze';stars=1;amt=1;}

  state.materials.wood+=amt;state.materials.stone+=amt;state.materials.iron+=amt;

  state.totalStars+=stars;

  BUILDINGS[state.currentBuilding].built=true;

  BUILDINGS[state.currentBuilding].quality=quality;

  const qNames={gold:'Gold',silver:'Silver',bronze:'Bronze'};

  const qColors={gold:'#FFD700',silver:'#C0C0C0',bronze:'#CD7F32'};

  document.getElementById('reward-icon').textContent=BUILDINGS[state.currentBuilding].icon;

  document.getElementById('reward-stars').innerHTML=[0,1,2].map(i=>`<span style="color:${i<stars?'#FFD700':'rgba(255,255,255,.15)'}">\u2B50</span>`).join('');

  document.getElementById('reward-title').textContent=BUILDINGS[state.currentBuilding].name+' Built!';

  document.getElementById('reward-title').style.color=qColors[quality];

  const isNew=wpm>=state.personalBestWPM;

  document.getElementById('reward-detail').innerHTML=

    `${qNames[quality]} quality! You read at <strong>${wpm} words per minute</strong>.<br>`+

    (isNew?'\u{1F389} New personal best!':`Your best is ${state.personalBestWPM} WPM. Try again to beat it!`);

  document.getElementById('reward-materials').innerHTML=

    `<div class="reward-material">\u{1FA93}<span>+${amt}</span></div>`+

    `<div class="reward-material">\u26CF\uFE0F<span>+${amt}</span></div>`+

    `<div class="reward-material">\u{1F528}<span>+${amt}</span></div>`;

  showScreen('reward-screen');launchConfetti();saveState();

}
