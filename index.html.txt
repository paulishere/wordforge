function shuffle(a){const b=[...a];for(let i=b.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[b[i],b[j]]=[b[j],b[i]];}return b;}
function clamp(v,lo,hi){return Math.max(lo,Math.min(hi,v));}

function renderBuildHeader(containerId, stage){
  const c=document.getElementById(containerId);
  const b=BLDGS[S.curBldg];
  const pct=S.buildProgress;
  const qClass=pct>=80?'q-gold':pct>=50?'q-silver':'q-bronze';
  const qLabel=pct>=80?'Gold':pct>=50?'Silver':'Bronze';
  // Icon scale based on progress
  const iconScale=0.5+((pct/100)*0.5);
  c.innerHTML=`
    <div class="bh-icon" id="bh-icon" style="transform:scale(${iconScale});opacity:${0.4+pct/166}">${b.icon}</div>
    <div class="bh-right">
      <div class="bh-label"><span class="bh-label-name">${b.name}</span><span>${qLabel} · ${Math.round(pct)}%</span></div>
      <div class="bh-bar">
        <div class="bh-fill ${qClass}" style="width:${pct}%"></div>

        ${[0,1,2].map(i=>`<div class="bh-dot ${i===stage?'active':i<stage?'done':''}"></div>`).join('')}

// ======== CHALLENGE FLOW ========
let dwQueue=[],dwIdx=0;

function startChallenge(i){
  S.curBldg=i;S.curChallenge=pickChallenge(i);S.buildProgress=0;S.compAttempts=0;
  dwQueue=[...S.curChallenge.decodeWords];dwIdx=0;showDecodeWord();
}
function replayChallenge(){
  S.curChallenge=pickChallenge(S.curBldg);S.buildProgress=0;S.compAttempts=0;
  dwQueue=[...S.curChallenge.decodeWords];dwIdx=0;showDecodeWord();
}

// ======== DECODE ========
// Each correct letter: +30/(total letters across all words)
// Each wrong letter: -4
function calcDecodeLetterValue(){
  const total=dwQueue.reduce((s,w)=>s+w.length,0);
  return 30/total;
}

function showDecodeWord(){
  const word=dwQueue[dwIdx];S.decodeTarget=word.toLowerCase();S.decodeLIdx=0;
  renderBuildHeader('decode-bh',0);
  const anvil=document.getElementById('anvil-area');anvil.innerHTML='';
  for(let i=0;i<word.length;i++){const s=document.createElement('div');s.className='anvil-slot';s.id='slot-'+i;anvil.appendChild(s);}
  const tray=document.getElementById('letter-tray');tray.innerHTML='';
  shuffle(word.split('')).forEach((ch,i)=>{const t=document.createElement('div');t.className='letter-tile';t.textContent=ch.toLowerCase();t.dataset.letter=ch.toLowerCase();t.dataset.index=i;t.onclick=()=>handleLetter(t);tray.appendChild(t);});
  document.getElementById('decode-instruction').textContent='Listen, then tap letters in order.'+(dwQueue.length>1?` Word ${dwIdx+1} of ${dwQueue.length}`:'');
  show('decode-screen');setTimeout(()=>speakCurrentWord(),500);
}

function handleLetter(tile){
  const expected=S.decodeTarget[S.decodeLIdx];
  if(tile.dataset.letter===expected){
    tile.classList.add('used');
    const slot=document.getElementById('slot-'+S.decodeLIdx);slot.textContent=tile.dataset.letter;slot.classList.add('filled');
    S.decodeLIdx++;
    addProgress(calcDecodeLetterValue());
    renderBuildHeader('decode-bh',0);animateBuildIcon('grow');
    if(S.decodeLIdx===S.decodeTarget.length){
      for(let i=0;i<S.decodeTarget.length;i++)document.getElementById('slot-'+i).classList.add('correct-flash');
      S.streak++;if(S.streak>S.bestStreak)S.bestStreak=S.streak;
      dwIdx++;
      if(dwIdx<dwQueue.length)setTimeout(()=>showDecodeWord(),1000);
      else setTimeout(()=>showRead(),1000);
    }
  }else{
    tile.style.pointerEvents='none';
    const slot=document.getElementById('slot-'+S.decodeLIdx);slot.classList.add('wrong-flash');
    removeProgress(4);renderBuildHeader('decode-bh',0);animateBuildIcon('shrink');
    setTimeout(()=>{slot.classList.remove('wrong-flash');tile.style.pointerEvents='auto';},500);
  }
}

  document.getElementById('forge-fill').style.background='linear-gradient(90deg,#CD7F32,#e94560,#FFD700)';
  document.getElementById('forge-fill').style.filter='brightness(1)';
  document.getElementById('forge-label').textContent='Tap the first word to begin!';
  show('read-screen');
}

function handleWord(index){
  if(index!==S.readIdx)return;
  const now=Date.now();const words=document.querySelectorAll('.word-token');
  if(index===0)S.readStart=now;
  S.readTimes.push(now);
  words[index].classList.remove('current');
  if(S.readMissedWords.has(index)){words[index].classList.add('missed');}
  else{words[index].classList.add('read');}
  const total=words.length;
  const pct=((index+1)/total)*100;
  document.getElementById('forge-fill').style.width=pct+'%';
  // Add build progress for each word
  if(!S.readMissedWords.has(index)){
    addProgress(30/total);renderBuildHeader('read-bh',1);animateBuildIcon('grow');
  }
  if(index>2){const elapsed=(now-S.readStart)/60000;const wpm=(index+1)/elapsed;updateForge(wpm);}
  S.readIdx++;
  if(S.readIdx<total){words[S.readIdx].classList.remove('unread');words[S.readIdx].classList.add('current');}
  else{setTimeout(()=>showComp(),600);}
}

function logMiss(){
  if(S.readIdx<=0)return;
  // Mark the current word as missed (the one being read now, or just read)
  const idx=S.readIdx>0?S.readIdx-1:0;
  if(S.readMissedWords.has(idx))return;
  S.readMisses++;S.readMissedWords.add(idx);
  document.getElementById('miss-count').textContent=S.readMisses;
  // Mark the word visually
  const words=document.querySelectorAll('.word-token');
  if(words[idx]){words[idx].classList.remove('read');words[idx].classList.add('missed');}
  removeProgress(3);renderBuildHeader('read-bh',1);animateBuildIcon('shrink');
}

function updateForge(wpm){
  const label=document.getElementById('forge-label');const fill=document.getElementById('forge-fill');
  const baseline=S.personalBestWPM>0?S.personalBestWPM:70;
  if(wpm>=baseline*1.1){label.textContent='\u{1F525} BLAZING! The forge is white-hot!';fill.style.background='linear-gradient(90deg,#FFD700,#fff,#FFD700)';fill.style.filter='brightness(1.3)';}
  else if(wpm>=baseline*0.85){label.textContent='\u{1F525} Hot! The forge glows bright!';fill.style.background='linear-gradient(90deg,#e94560,#FFD700)';fill.style.filter='brightness(1.1)';}
  else{label.textContent='\u2728 Warming up... keep going!';fill.style.background='linear-gradient(90deg,#CD7F32,#e94560)';fill.style.filter='brightness(1)';}
}

// ======== COMPREHEND ========
// Correct first try: +40  Correct second try: +20  Each wrong: -10
function showComp(){
  const c=S.curChallenge;S.compAttempts=0;
  renderBuildHeader('comp-bh',2);
  document.getElementById('comp-question').textContent=c.question;
  const opt=document.getElementById('comp-options');opt.innerHTML='';
  c.answers.forEach((a,i)=>{const btn=document.createElement('div');btn.className='comp-option';btn.textContent=a;btn.onclick=()=>handleComp(i,btn);opt.appendChild(btn);});
  show('comprehend-screen');
}

function handleComp(index,btn){
  const c=S.curChallenge;
  const all=document.querySelectorAll('.comp-option');
  all.forEach(b=>b.style.pointerEvents='none');
  S.compAttempts++;
  if(index===c.correctAnswer){
    btn.classList.add('correct');S.streak++;if(S.streak>S.bestStreak)S.bestStreak=S.streak;
    if(S.compAttempts===1){addProgress(40);animateBuildIcon('big-grow');}
    else{addProgress(20);animateBuildIcon('grow');}
    renderBuildHeader('comp-bh',2);
    setTimeout(()=>showReward(),800);
  }else{
    btn.classList.add('incorrect');S.streak=0;
    removeProgress(10);renderBuildHeader('comp-bh',2);animateBuildIcon('shrink');
    setTimeout(()=>{all.forEach(b=>{b.classList.remove('correct','incorrect');b.style.pointerEvents='auto';});},1200);
  }
}

function showReward(){
  const words=S.curChallenge.passage.split(/\s+/);
  const totalTime=(S.readTimes[S.readTimes.length-1]-S.readStart)/60000;
  const wpm=Math.round(words.length/totalTime);
  if(wpm>S.personalBestWPM)S.personalBestWPM=wpm;

  const pct=S.buildProgress;
  let quality,stars,amt;
  if(pct>=80){quality='gold';stars=3;amt=3;}
  else if(pct>=50){quality='silver';stars=2;amt=2;}
  else{quality='bronze';stars=1;amt=1;}

  S.materials.wood+=amt;S.materials.stone+=amt;S.materials.iron+=amt;
  S.totalStars+=stars;
  BLDGS[S.curBldg].built=true;BLDGS[S.curBldg].quality=quality;
  const qN={gold:'Gold',silver:'Silver',bronze:'Bronze'};
  const qC={gold:'#FFD700',silver:'#C0C0C0',bronze:'#CD7F32'};
  document.getElementById('reward-icon').textContent=BLDGS[S.curBldg].icon;
  document.getElementById('reward-stars').innerHTML=[0,1,2].map(i=>`<span style="color:${i<stars?'#FFD700':'rgba(255,255,255,.15)'}">\u2B50</span>`).join('');
  document.getElementById('reward-title').textContent=BLDGS[S.curBldg].name+' Built!';
  document.getElementById('reward-title').style.color=qC[quality];

  let detail=`${qN[quality]} quality! You read at <strong>${wpm} WPM</strong>.`;
  if(S.readMisses>0)detail+=`<br>${S.readMisses} word${S.readMisses>1?'s':''} missed during reading.`;
  if(S.compAttempts===1)detail+=`<br>\u{1F389} Comprehension correct on the first try!`;
  else detail+=`<br>Comprehension took ${S.compAttempts} attempts.`;

  document.getElementById('reward-detail').innerHTML=detail;
  document.getElementById('reward-materials').innerHTML=
    `<div class="reward-material">\u{1FA93}<span>+${amt}</span></div>`+
    `<div class="reward-material">\u26CF\uFE0F<span>+${amt}</span></div>`+
    `<div class="reward-material">\u{1F528}<span>+${amt}</span></div>`;
  show('reward-screen');launchConfetti();save();
}
